> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Implement full AT Protocol Sync 1.1 verification and verifier-aware streaming in shrike.

**Architecture:** Add raw sync event parsing, an async caller-provided state store trait, a verifier with typed errors/stats/policies, MST inversion and resync support, async resync workers, and verifier-aware streaming with per-DID FIFO and cross-DID parallelism. Keep no-verifier streaming behavior compatible and keep persistence behind `StateStore`.

**Tech Stack:** Rust 2024, Tokio, futures, async-trait, existing shrike `syntax`, `cbor`, `car`, `mst`, `repo`, `identity`, `xrpc`, and `streaming` modules.

use std::collections::HashMap;
use tokio::sync::RwLock;

use crate::cbor::Cid;
use crate::syntax::Did;
use crate::sync::SyncError;

#[derive(Debug, Clone, PartialEq, Eq, serde::Serialize, serde::Deserialize)]
pub struct ChainState {
    pub rev: String,
    pub data: Cid,
}

#[derive(Debug, Clone, PartialEq, Eq, serde::Serialize, serde::Deserialize)]
pub struct HostingState {
    pub active: bool,
    pub status: Option<String>,
    pub seq: i64,
    pub time: String,
}

impl HostingState {
    pub fn is_active(state: Option<&Self>) -> bool {
        state.map(|s| s.active).unwrap_or(true)
    }
}

#[async_trait::async_trait]
pub trait StateStore: Send + Sync {
    async fn load_chain(&self, did: &Did) -> Result<Option<ChainState>, SyncError>;
    async fn save_chain(&self, did: &Did, state: ChainState) -> Result<(), SyncError>;
    async fn load_hosting(&self, did: &Did) -> Result<Option<HostingState>, SyncError>;
    async fn save_hosting(&self, did: &Did, state: HostingState) -> Result<(), SyncError>;
    async fn delete(&self, did: &Did) -> Result<(), SyncError>;
}

#[derive(Default)]
pub struct MemStateStore {
    chain: RwLock<HashMap<Did, ChainState>>,
    hosting: RwLock<HashMap<Did, HostingState>>,
}

impl MemStateStore {
    pub fn new() -> Self {
        Self::default()
    }
}

#[async_trait::async_trait]
impl StateStore for MemStateStore {
    async fn load_chain(&self, did: &Did) -> Result<Option<ChainState>, SyncError> {
        Ok(self.chain.read().await.get(did).cloned())
    }

    async fn save_chain(&self, did: &Did, state: ChainState) -> Result<(), SyncError> {
        self.chain.write().await.insert(did.clone(), state);
        Ok(())
    }

    async fn load_hosting(&self, did: &Did) -> Result<Option<HostingState>, SyncError> {
        Ok(self.hosting.read().await.get(did).cloned())
    }

    async fn save_hosting(&self, did: &Did, state: HostingState) -> Result<(), SyncError> {
        self.hosting.write().await.insert(did.clone(), state);
        Ok(())
    }

    async fn delete(&self, did: &Did) -> Result<(), SyncError> {
        self.chain.write().await.remove(did);
        self.hosting.write().await.remove(did);
        Ok(())
    }
}
```

pub fn find_duplicate_path(ops: &[RawRepoOp]) -> Option<&str> {
    let mut seen = HashSet::with_capacity(ops.len());
    for op in ops {
        if !seen.insert(op.path.as_str()) {
            return Some(op.path.as_str());
        }
    }
    None
}

pub fn invert_commit(raw: &RawCommit) -> Result<Cid, VerifierError> {
    if let Some(path) = find_duplicate_path(&raw.ops) {
        return Err(VerifierError::DuplicatePath { did: raw.repo.clone(), rev: raw.rev.to_string(), path: path.to_owned() });
    }
    let decoded = decode_commit_car(raw)?;
    let mut tree = Tree::load(Box::new(decoded.store.clone()), decoded.inner.data);
    for op in &raw.ops {
        match op.action.as_str() {
            "create" => {
                tree.remove(&op.path)?;
            }
            "update" | "delete" => {
                let prev = op.prev.ok_or_else(|| VerifierError::Inversion {
                    did: raw.repo.clone(),
                    rev: raw.rev.to_string(),
                    message: format!("{} op {:?} missing prev CID", op.action, op.path),
                })?;
                tree.insert(op.path.clone(), prev)?;
            }
            other => {
                return Err(VerifierError::Inversion {
                    did: raw.repo.clone(),
                    rev: raw.rev.to_string(),
                    message: format!("unknown op action {other:?}"),
                });
            }
        }
    }
    Ok(tree.root_cid()?)
}
```
